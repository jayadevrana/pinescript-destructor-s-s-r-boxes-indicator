// Helpers to color a box based on previous box bounds
f_apply_color_for_bounds(b, thisLow, thisHigh) =>
    float prevTop = na
    float prevBot = na
    int sz = array.size(boxes)
    if sz >= 2
        box prev = array.get(boxes, sz - 2)
        prevTop := box.get_top(prev)
        prevBot := box.get_bottom(prev)
    color edgeCol = colNeutralInput
    color bgCol   = color.new(colNeutralInput, fillTransp)
    if not na(prevTop) and not na(prevBot)
        if thisLow > prevTop
            edgeCol := colUpInput
            bgCol   := color.new(colUpInput, fillTransp)
        else if thisHigh < prevBot
            edgeCol := colDownInput
            bgCol   := color.new(colDownInput, fillTransp)
    box.set_border_color(b, edgeCol)
    box.set_bgcolor(b, bgCol)

hig_h = request.security(syminfo.tickerid,"D",high)
lo_w = request.security(syminfo.tickerid,"D",low)

// Compute pre-window extremes and left time (bar_time)
f_prewindow() =>
    float preHi = high[1]
    float preLo = low[1]
    if pre_candle_from_e > 0
        for i = 2 to pre_candle_from_e + 1
            preHi := math.max(preHi, high[i])
            preLo := math.min(preLo, low[i])
    int lTime = time[pre_candle_from_e + 1]
    [preHi, preLo, lTime]

// Cap the number of boxes (delete oldest)
f_enforce_limit() =>
    while array.size(boxes) > number_of_boxes
        box toDel = array.shift(boxes)
        box.delete(toDel)

// ─────────────────────── Earnings trigger ───────────────────────

if isE1 and bar_index > pre_candle_from_e + 1
    if extend_to_next and array.size(boxes) > 0
        box lastB = array.get(boxes, array.size(boxes) - 1)
        box.set_right(lastB, time)

    [preHi, preLo, lTime] = f_prewindow()
    leftTime := lTime

    box b = box.new(left = leftTime, top = preHi, right = time, bottom = preLo, xloc = xloc.bar_time, border_style = line.style_dashed, bgcolor = color.new(colNeutralInput, fillTransp), border_color = colNeutralInput)
    array.push(boxes, b)

    accHigh := preHi
    accLow  := preLo
    forming := true
    barsLeft := math.max(0, post_candle_from_e)

    accHigh := math.max(accHigh, high)
    accLow  := math.min(accLow,  low)
    box.set_top(b, accHigh)
    box.set_bottom(b, accLow)
    box.set_right(b, time)
    f_apply_color_for_bounds(b, accLow, accHigh)
    barsLeft := math.max(0, barsLeft - 1)

    box cur = array.get(boxes, array.size(boxes) - 1)
    if barsLeft > 0
        accHigh := math.max(accHigh, high)
        accLow  := math.min(accLow,  low)
        box.set_top(cur, accHigh)
        box.set_bottom(cur, accLow)
        f_apply_color_for_bounds(cur, accLow, accHigh)
        barsLeft -= 1
        box.set_right(cur, time)
    else
        forming := false
